def _ensure_pool(self) -> multiprocessing.pool.Pool:
    """Lazily create and return a persistent worker pool.

    Uses pickled FE state to initialize workers (~1ms per worker vs ~30ms
    loading from disk).  Falls back to disk-based init on pickle failure.

    Always uses spawn context for cross-platform consistency (Windows
    requires spawn; Linux defaults changed in Python 3.14).
    """
    if self._pool is None:
        workers = self._effective_workers()
        ctx = multiprocessing.get_context("spawn")
        try:
            fe_state = pickle.dumps(self.feature_engineer)
            self._pool = ctx.Pool(
                processes=workers,
                initializer=_init_fe_worker_pickled,
                initargs=(fe_state,),
            )
        except (pickle.PicklingError, TypeError) as exc:
            _log.warning(
                "FeatureEngineer pickle failed (%s); falling back to ~30x slower disk-based "
                "worker init (~30ms per worker vs ~1ms). Fix: ensure FeatureEngineer is "
                "picklable, or check __getstate__/__setstate__.",
                exc,
            )
            self._pool = ctx.Pool(
                processes=workers,
                initializer=_init_fe_worker,
                initargs=(self.feature_engineer.config, self.feature_engineer.repo_root),
            )
    return self._pool
```

The only change is:
1. Add `ctx = multiprocessing.get_context("spawn")` before the try block
2. Replace `multiprocessing.Pool(` with `ctx.Pool(` in both the try and except branches

Everything else (initargs, initializer functions, fallback logic) stays identical.

class TestDataLoaderSpawnContext:
    """Tests for DataLoader spawn context (GLBL-02)."""

    def test_ensure_pool_uses_spawn_context(self, monkeypatch):
        """DataLoader._ensure_pool() creates pool via mp.get_context('spawn')."""
        import multiprocessing
        from unittest.mock import MagicMock, patch

        calls = []
        original_get_context = multiprocessing.get_context

        def tracking_get_context(method):
            calls.append(method)
            return original_get_context(method)

        monkeypatch.setattr(multiprocessing, "get_context", tracking_get_context)

        # Create a DataLoader with a small worker count
        # Use tmp_path-like dir to avoid needing real data
        dl = DataLoader(data_dir=".", n_workers=1)

        # Mock the pool creation to avoid actually spawning processes
        mock_pool = MagicMock()
        ctx = multiprocessing.get_context("spawn")

        with patch.object(type(ctx), "Pool", return_value=mock_pool):
            dl._ensure_pool()

        # Verify get_context was called with "spawn"
        assert "spawn" in calls, (
            f"Expected get_context('spawn') to be called, got calls: {calls}"
        )
        # Clean up
        dl._pool = None

    def test_worker_init_via_initargs(self):
        """DataLoader passes FeatureEngineer state through initargs, not globals."""
        import inspect
        source = inspect.getsource(DataLoader._ensure_pool)
        # Must use initargs parameter (already does, this is a regression guard)
        assert "initargs" in source, "Pool must use initargs for worker initialization"
        # Must NOT mutate module-level _worker_fe directly in _ensure_pool
        assert "global _worker_fe" not in source, (
            "_ensure_pool must not mutate module-level _worker_fe directly"
        )
```
